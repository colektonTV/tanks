import arcade

WIDTH = 800
HEIGHT = 600
SPEED = 5
SIZE = 40


class App(arcade.Window):
    def __init__(self):
        super().__init__(WIDTH, HEIGHT)
        arcade.set_background_color(arcade.color.AMAZON)
        self.x = WIDTH // 2
        self.y = HEIGHT // 2
        self.keys = set()

    def on_draw(self):
        self.clear()
        arcade.draw_rectangle_filled(self.x, self.y, SIZE, SIZE, arcade.color.GRAY)
        arcade.draw_rectangle_filled(self.x, self.y + 10, 8, 30, arcade.color.BLACK)

    def update(self, delta):
        dx = 0
        dy = 0

        if {arcade.key.W, arcade.key.UP} & self.keys:
            dy = SPEED
        if {arcade.key.S, arcade.key.DOWN} & self.keys:
            dy = -SPEED
        if {arcade.key.A, arcade.key.LEFT} & self.keys:
            dx = -SPEED
        if {arcade.key.D, arcade.key.RIGHT} & self.keys:
            dx = SPEED

        self.x += dx
        self.y += dy

        limit = SIZE / 2
        self.x = max(limit, min(self.x, WIDTH - limit))
        self.y = max(limit, min(self.y, HEIGHT - limit))

    def on_key_press(self, key, mods):
        self.keys.add(key)

    def on_key_release(self, key, mods):
        self.keys.discard(key)


if __name__ == "__main__":
    app = App()
    arcade.run()
